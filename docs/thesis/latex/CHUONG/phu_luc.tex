\chapter{BẢN GHI LƯỢT CHẤM}
\label{appendix:ban_ghi}

Bảng~\ref{tab:ban_ghi} là bố cục bản ghi 48~B trong nhật ký lượt chấm. Bố cục này khai ở
tệp \mbox{\texttt{storage\_format.h}} của thành phần \texttt{sys\_storage}.

\begin{table}[!htb]
\centering
\small
\caption{Bố cục bản ghi lượt chấm.}
\label{tab:ban_ghi}
\begin{tabular}{|c|c|L{3.8cm}|L{6.4cm}|}
\hline
\textbf{Vị trí} & \textbf{Cỡ} & \textbf{Trường} & \textbf{Ý nghĩa} \\
\hline
0 & 4 & \texttt{magic} & Hằng \texttt{ATTD} \\
\hline
4 & 8 & \texttt{local\_id} & Số lần khởi động ghép số thứ tự trong phiên \\
\hline
12 & 4 & \texttt{employee\_id} & Mã nhân viên do máy chủ cấp \\
\hline
16 & 8 & \texttt{ts\_ms} & Giờ chấm, mili giây từ epoch \\
\hline
24 & 1 & \texttt{direction} & 0 là vào, 1 là ra \\
\hline
25 & 2 & \texttt{match\_score} & Điểm so khớp, dạng Q8.8 \\
\hline
27 & 2 & \texttt{liveness\_score} & Điểm sống, dạng Q8.8 \\
\hline
29 & 1 & \texttt{flags} & Đã mở thanh chắn, thu lúc mất mạng, giờ chưa đồng bộ NTP \\

\hline
32 & 12 & \texttt{reserved} & Để dành \\
\hline
44 & 4 & \texttt{crc32} & Kiểm tra toàn vẹn \\
\hline
\end{tabular}
\end{table}
